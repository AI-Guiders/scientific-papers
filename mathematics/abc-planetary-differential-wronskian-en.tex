\begin{abstract}
We formulate any coprime additive triple $A + B = C$ ($A = a_1 a_2,\ B = b_1 b_2,\ C = c_1 c_2$, with $k$-th floor decomposition $A = \widetilde{A}_k X_k^k,\ B = \widetilde{B}_k Y_k^k,\ C = \widetilde{C}_k Z_k^k$ and root weights $\widetilde{A}_k\widetilde{B}_k\widetilde{C}_k \le \operatorname{rad}(ABC)^{k-1}$) as a three-shaft planetary gear differential with orthogonal $3\mathrm{D}$ frame $\mathcal{F}_3 = (\mathbf{r},\mathbf{\omega},\mathbf{r}\times\mathbf{\omega})\in\mathbb{Z}^{3\times 3}$ ($\mathbf{r} = (a_1,b_1,c_1)\perp\mathbf{\omega} = (a_2,b_2,-c_2)$), planar gear-slip Brahmagupta law $C^2 + S_{AB}^2 = (A-B)^2 + M_{AB}^2 = (a_1^2+b_1^2)(a_2^2+b_2^2)$, non-singular Hankel--Vandermonde moment matrix $-\det\mathcal{H}_3 = \widetilde{A}\widetilde{B}\widetilde{C}(X-Y)^2(Y-Z)^2(Z-X)^2 > 0$, Lagrange--Vandermonde shaft-gap projectors, and unimodular $2\times 3$ conjugate B\'ezout partner matrix $\mathcal{B}_{2\times 3}$ (\texttt{thm8}, \texttt{thm9}). We then prove five standalone universally $\forall$-quantified Diophantine and geometric theorems across $\mathbb{Z}$: reverse Pad\'e--B\'ezout cubic annihilation and quadratic-shaft descent $I_2$ (\texttt{thm12}); the integer Mason--Stothers--B\'ezout cubic Wronskian $W^* = A B^* - B A^* \equiv X^2 Y^2 Z^2$ with fourth-power determinant $-\det\mathcal{W}_{\mathrm{MS}} \equiv X^4 + Y^4 + Z^4$ (\texttt{thm13}); unconditional circle-remainder degree reduction with the full radical-shaft skeleton $R_{\mathrm{full}} = (\widetilde{A}\widetilde{B}\widetilde{C})(X Y Z)$ (\texttt{thm14}); cross-floor $k=2\leftrightarrow k=3$ coupling onto the 6th-degree Fermat--Selmer curve of genus $g(6)=10\ge 2$ (\texttt{thm15}); and universal Weierstrass stereographic embedding into an arbitrary four-quadrant rotator ring $\mathbb{Z}_{4Q}$ ($\forall Q\in\mathbb{Z}$, recovering $\mathbb{Z}_{512}$ at $Q=128$) with quarter-modulus invariant $C\Delta_{\mathrm{rot},Q} - w I_{\mathrm{rot},Q} \equiv Q$ (\texttt{thm16}). All theorems are verified in \texttt{AbcPlanetaryWronskian.lean} (\texttt{0 sorryAx}, \texttt{0 Classical.choice}).

\medskip\noindent
\textbf{Keywords:} additive Diophantine triples, $abc$-triples, Mason--Stothers Wronskian, Pad\'e--B\'ezout annihilation, Fermat--Selmer curves, Weierstrass stereographic homeomorphism, Lean~4.

\smallskip\noindent
\textbf{MSC 2020:} 11D41, 11J86, 14H45, 68V20.
\end{abstract}

\section{Planetary Differential Frame \texorpdfstring{$\mathcal{F}_3$}{F3}, Hankel--Vandermonde Matrix \texorpdfstring{$\mathcal{H}_3$}{H3}, and Three-Channel Pad\'e Matrix \texorpdfstring{$\mathcal{P}_3$}{P3}}
\label{sec:planetary-frame}

Following A.\,I.~Mal'cev's model-theoretic terminology (\emph{Algebraic Systems}~\cite{Malcev1973}, where the \emph{carrier} / \emph{носитель} is the underlying domain set of an algebraic sort), we treat the rotator system as a two-sorted algebraic structure $\langle\mathcal{M},\mathcal{T};\,\pi_{M\to T},\,\iota_{T\to M}\rangle$ linking the \textbf{discrete $M$-carrier} ($\mathbb{Z}_{4Q}$ with $L^1$ projections $\mathcal{C}_{4Q},\mathcal{S}_{4Q}$) and the \textbf{continuous Euclidean $T$-carrier} ($L^2$ circle $X^2+Y^2=R^2$ and continuum line $\mathbb{R}$).

\begin{theorem}[Radical-bounded floor embedding, bordered $3\times 3$ matrix $\mathcal{M}_3$, orthogonal planetary differential frame $\mathcal{F}_3$, and Hankel--Vandermonde matrix $\mathcal{H}_3$, \texttt{thm8}]
\label{thm:main-8}
For any additive triple $A + B = C$ ($A = a_1 a_2,\ B = b_1 b_2,\ C = c_1 c_2$), the bordered matrix $\mathcal{M}_3 = \begin{pmatrix} c_2 & a_2 & b_2 \\ a_1 & c_1 & 0 \\ b_1 & 0 & c_1 \end{pmatrix}$ satisfies $\det\mathcal{M}_3 = c_1(C-A-B) = 0$ and rank-$1$ adjugate idempotence $(\mathcal{M}_3^{\mathrm{adj}})^2 = (c_1^2+C)\mathcal{M}_3^{\mathrm{adj}}$. The gear-radius vector $\mathbf{r} = (a_1,b_1,c_1)$ and turn-count vector $\mathbf{\omega} = (a_2,b_2,-c_2)$ form an orthogonal frame $\mathcal{F}_3 = (\mathbf{r},\mathbf{\omega},\mathbf{r}\times\mathbf{\omega})$ with $\mathbf{r}\cdot\mathbf{\omega} = A+B-C = 0$, $\det\mathcal{F}_3 = M_{BC}^2 + M_{CA}^2 + S_{AB}^2 \equiv \|\mathbf{r}\|_2^2\|\mathbf{\omega}\|_2^2$, and planar Brahmagupta slip law $C^2 + S_{AB}^2 = (A-B)^2 + M_{AB}^2 \equiv (a_1^2+b_1^2)(a_2^2+b_2^2)$. On floor $k=3$ ($A = \widetilde{A}X^3,\ B = \widetilde{B}Y^3,\ C = \widetilde{C}Z^3$, $S_m = \widetilde{A}X^m + \widetilde{B}Y^m - \widetilde{C}Z^m$, $S_3 = 0$), the Hankel moment matrix $\mathcal{H}_3 = \begin{pmatrix} S_0 & S_1 & S_2 \\ S_1 & S_2 & 0 \\ S_2 & 0 & S_4 \end{pmatrix}$ satisfies $-\det\mathcal{H}_3 \equiv \widetilde{A}\widetilde{B}\widetilde{C}(X-Y)^2(Y-Z)^2(Z-X)^2$ and the Lagrange--Vandermonde shaft-gap projectors $\widetilde{A}X(X-Y)(X-Z) \equiv S_1 YZ - S_2(Y+Z)$, $\widetilde{B}Y(Y-X)(Y-Z) \equiv S_1 ZX - S_2(Z+X)$, and $\widetilde{C}Z(Z-X)(Z-Y) \equiv S_2(X+Y) - S_1 XY$.
\end{theorem}

\begin{theorem}[Unimodular $2\times 3$ conjugate B\'ezout partner matrix $\mathcal{B}_{2\times 3}$ and three-channel cubic Pad\'e matrix $\mathcal{P}_3(A,B,C)$, \texttt{thm9}]
\label{thm:main-9}
For any $A + B = C$ and any B\'ezout pair $A v - B u = 1$ ($w = u + v$), the $2\times 3$ conjugate matrix $\mathcal{B}_{2\times 3} = \begin{pmatrix} A & B & C \\ u & v & w \end{pmatrix}$ has all three $2\times 2$ minors equal to $+1$ ($Av-Bu = Aw-Cu = Bw-Cv = 1$), while the three-channel cubic Pad\'e matrix $\mathcal{P}_3(A,B,C)$ has exact null-vector $(A,B,-C)^T$ ($\mathcal{P}_3(A,B,C)(A,B,-C)^T = \mathbf{0}$) and trace $\operatorname{Tr}\mathcal{P}_3(A,B,C) = A C(A+C)$.
\end{theorem}

\section{Universal \texorpdfstring{$\forall$}{For-All}-Quantified Mason--Stothers Wronskian, Circle-Remainder Reduction, Genus-10 Lock, and \texorpdfstring{$\mathbb{Z}_{4Q}$}{Z\_4Q} Stereographic Embedding}
\label{sec:universal-theorems}

\begin{theorem}[Reverse Pad\'e--B\'ezout cubic annihilation, Stern--Brocot mediant elimination, and quadratic-shaft descent $I_2$, \texttt{thm12}]
\label{thm:reverse-pade-thm12}
For $A = \widetilde{C}Z^3 - \widetilde{B}Y^3$, $I_0 = Yw - Zv$, and $I_1 = Yw(\widetilde{C}w^3+2\widetilde{B}v^3) - Zv(2\widetilde{C}w^3+\widetilde{B}v^3)$,
\begin{equation}
  \label{eq:reverse-pade-bezout}
  Z^3 I_1 \;+\; A\,w^3(2Zv - Yw) \;\equiv\; \widetilde{B}\,I_0^3(Yw + Zv),\qquad
  Y^3 I_1 \;+\; A\,v^3(2Yw - Zv) \;\equiv\; \widetilde{C}\,I_0^3(Yw + Zv),
\end{equation}
and between the left and right Stern--Brocot parents $(v,w)$ and $(v_m,w_m) = (Y-v,Z-w)$ of the Farey mediant $(Y,Z)$ ($p = v_m w,\ q = v w_m,\ K_+ = w^3(2Zv-Yw),\ K_- = w_m^3(2Zv_m-Yw_m)$), eliminating $A$ cancels $Z^3$ identically: $I_1 K_- - I_{1m} K_+ \equiv \widetilde{B} I_0^3\bigl((vw + v_m w_m)(p+q) - (p^2+q^2)\bigr)$. Moreover, the quadratic-shaft descent invariant $I_2 := \widetilde{B}Yv^2(3Yw-Zv) - \widetilde{C}Zw^2(3Zv-Yw)$ eliminates $(v,w)$ from the right-hand side ($Z^2 I_2 + A w^2(3Zv-Yw) \equiv \widetilde{B}Y I_0^3$, $Y^2 I_2 + A v^2(3Yw-Zv) \equiv \widetilde{C}Z I_0^3$) and forms an $A$-free $2\times 2$ Wronskian system with $I_1$ of determinant $3 I_0^3$.
\end{theorem}

\begin{theorem}[Integer Mason--Stothers--B\'ezout cubic Wronskian and fourth-power determinant, \texttt{thm13}]
\label{thm:mason-stothers-thm13}
For $\widetilde{A}X^3 + \widetilde{B}Y^3 = \widetilde{C}Z^3$ and $\widetilde{A}Xv_0 - \widetilde{B}Yu_0 = 1$, setting $\alpha = u_0 Z^2 + t\widetilde{A}X$, $\beta = v_0 Z^2 + t\widetilde{B}Y$, $\gamma = u_0 X^2 + v_0 Y^2 + t\widetilde{C}Z$ yields $A^*+B^*=C^*$ ($A^*=\alpha X^2,\ B^*=\beta Y^2,\ C^*=\gamma Z^2$) with shaft-square minors $(\widetilde{A}X\beta-\widetilde{B}Y\alpha,\ \widetilde{C}Z\alpha-\widetilde{A}X\gamma,\ \widetilde{C}Z\beta-\widetilde{B}Y\gamma) = (Z^2,Y^2,X^2)$ and exact Wronskian
\begin{equation}
  \label{eq:ms-wronskian}
  W^* \;:=\; A B^* - B A^* \;\equiv\; X^2 Y^2 Z^2,\qquad
  -\det\mathcal{W}_{\mathrm{MS}} \;\equiv\; X^4 + Y^4 + Z^4.
\end{equation}
\end{theorem}

\begin{theorem}[Circle-remainder full-skeleton degree reduction and $3\times 3$ cubic Cramer--B\'ezout lock, \texttt{thm14}]
\label{thm:circle-remainder-thm14}
With the full radical-shaft skeleton $R_{\mathrm{full}} := (\widetilde{A}\widetilde{B}\widetilde{C})(X Y Z)$, the volume identity $A B C \equiv R_{\mathrm{full}} W^*$ holds alongside
\begin{equation}
  \label{eq:circle-remainder-rfull}
  (\widetilde{B}Y)\,C \;+\; (\widetilde{B}^2\widetilde{C})(Y^2 Z)\,\alpha \;\equiv\; R_{\mathrm{full}}\,\beta,\qquad
  (\widetilde{A}X)\,\beta \;=\; Z^2 + (\widetilde{B}Y)\,\alpha.
\end{equation}
Reducing $u_0 Z^2$ on the circle of circumference $\widetilde{A}X$ to $0\le\alpha\le\widetilde{A}X-1$ yields $(\widetilde{A}X)\beta \le Z^2 + (\widetilde{B}Y)(\widetilde{A}X-1)$ and $(\widetilde{B}Y)C \le R_{\mathrm{full}}\beta$ (implying $C \le R_{\mathrm{full}}^2$ whenever $\beta\le R_{\mathrm{full}}$ and $C \le R_{\mathrm{full}}^3$ across all positive shafts and weights). For the $3\times 3$ cubic Cramer--B\'ezout system $U_0 Y^3 - V_0 X^3 = 1$ and $V_1 Z^3 - U_1 Y^3 = 1$, the determinant $\Delta_{\mathrm{cube}} := \widetilde{A}U_0 V_1 + \widetilde{B}V_0 V_1 - \widetilde{C}U_1 V_0 \in [1, \widetilde{A}+\widetilde{C}]$ satisfies $\Delta_{\mathrm{cube}}(X^3, Y^3, Z^3) = \bigl(\widetilde{C}(U_0+U_1)-\widetilde{B}V_1,\ \widetilde{A}V_1+\widetilde{C}V_0,\ \widetilde{A}(U_0+U_1)+\widetilde{B}V_0\bigr)$.
\end{theorem}

\begin{theorem}[Cross-floor $k=2 \leftrightarrow k=3$ coupling onto the genus-10 curve and Euler characteristic trichotomy, \texttt{thm15}]
\label{thm:genus-10-thm15}
The Euler characteristic of a degree-$k$ plane curve satisfies $\chi(\mathcal{C}_k) = 2 - (k-1)(k-2) \equiv k(3-k)$ (vanishing uniquely at $k = 3$, matching the simplicial Poincar\'e--Heegaard decomposition $S^3 = \mathcal{C}_{13}^{(+)}\cup_{O_2(3)}\mathcal{C}_{13}^{(-)}$ with $\chi(\mathcal{C}_{13}) = +1,\ \chi(O_2(3)) = +2,\ \chi(S^3) = 0$). Coupling the spherical floor $k=2$ ($A = A_2 X_2^2,\ B = B_2 Y_2^2,\ C = C_2 Z_2^2$, $\chi=+2$) and the toroidal floor $k=3$ ($A = A_3 X_3^3,\ B = B_3 Y_3^3,\ C = C_3 Z_3^3$, $\chi=0$) via $3-2=+1$ lifts $(U_6,V_6,W_6) = (X_2 Y_3 Z_3, Y_2 X_3 Z_3, Z_2 X_3 Y_3)$ onto the 6th-degree curve of genus $g(6) = 10 \ge 2$ ($\chi(\mathcal{C}_6) = -18 < 0$):
\begin{equation}
  \label{eq:genus-10-curve}
  A_2^3 B_3^2 C_3^2\,U_6^6 \;+\; B_2^3 A_3^2 C_3^2\,V_6^6 \;=\; C_2^3 A_3^2 B_3^2\,W_6^6.
\end{equation}
\end{theorem}

\begin{theorem}[Universal Weierstrass stereographic embedding into an arbitrary four-quadrant rotator $\mathbb{Z}_{4Q}$, \texttt{thm16}]
\label{thm:main-12-16}
For any $A+B=C\in\mathbb{Z}^3$ with B\'ezout partner $A v - B u = 1$ ($u+v=w$), the Weierstrass lift $X_T = A^2-B^2,\ Y_T = 2AB,\ R_T = A^2+B^2$ satisfies $X_T^2+Y_T^2=R_T^2$, $R_T+Y_T=C^2$, and $(C^2+X_T,C^2-X_T)=(2CA,2CB)$, while for any quarter-modulus $Q\in\mathbb{Z}$ on the four-quadrant ring $\mathbb{Z}_{4Q}$ (recovering $\mathbb{Z}_{512}$ at $Q = 128$) and any rotator cell $\varphi_0\in\mathbb{Z}$ ($\mathcal{C}_{4Q}(\varphi_0)=Q-\varphi_0,\ \mathcal{S}_{4Q}(\varphi_0)=\varphi_0,\ I_{\mathrm{rot},Q} = B\mathcal{C}_{4Q}(\varphi_0)-A\mathcal{S}_{4Q}(\varphi_0),\ \Delta_{\mathrm{rot},Q} = \mathcal{C}_{4Q}(\varphi_0)v-\mathcal{S}_{4Q}(\varphi_0)u$),
\begin{equation}
  \label{eq:quarter-ring-invariant}
  Q\,A \;=\; C\,\mathcal{C}_{4Q}(\varphi_0) - I_{\mathrm{rot},Q},\qquad
  Q\,B \;=\; C\,\mathcal{S}_{4Q}(\varphi_0) + I_{\mathrm{rot},Q},\qquad
  C\cdot\Delta_{\mathrm{rot},Q} \;-\; w\cdot I_{\mathrm{rot},Q} \;\equiv\; Q.
\end{equation}
\end{theorem}
